Integration Guide & React Component

Ready-to-use code snippets for integrating Streame into your Node.js or React app.

React / Next.js Component

import React from 'react';

export function VideoPlayer({ tmdbId, season, episode }) {
  const embedUrl = season && episode
    ? \`/embed/\${tmdbId}/\${season}/\${episode}\`
    : \`/embed/\${tmdbId}\`;

  return (
    <div className="relative w-full aspect-video rounded-xl overflow-hidden bg-black border border-zinc-800">
      <iframe
        src={embedUrl}
        className="w-full h-full border-0"
        allowFullScreen
        title="Streame Stream Player"
      />
    </div>
  );
}

Node.js / Express Setup

const express = require('express');
const app = express();

app.get('/embed/:tmdbId', (req, res) => {
  const { tmdbId } = req.params;
  const { server = 'barbarian', autoplay = '1' } = req.query;
  
  res.redirect(\`/embed/\${tmdbId}?server=\${server}&autoplay=\${autoplay}\`);
});

app.listen(3000, () => console.log('Streame node running on port 3000'));
